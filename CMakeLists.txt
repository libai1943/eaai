cmake_minimum_required(VERSION 3.16)
project(RegionCovOpt LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Eigen3 3.3 QUIET NO_MODULE)
if(NOT TARGET Eigen3::Eigen)
  add_library(Eigen3::Eigen INTERFACE IMPORTED)
  set_target_properties(Eigen3::Eigen PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_SOURCE_DIR}/population_motion_research_20261106/third_party/eigen3")
endif()
foreach(entry docking shared shoot_dock shoot_shared)
  if(entry STREQUAL "docking")
    set(binary dock)
  else()
    set(binary ${entry})
  endif()
  add_executable(${binary} eaai48/revision/src/${entry}.cpp)
  target_link_libraries(${binary} PRIVATE Eigen3::Eigen)
  target_compile_definitions(${binary} PRIVATE EAAI_ARM_OVERTRAVEL EAAI_SWEEP)
  set_target_properties(${binary} PROPERTIES SUFFIX ".exe" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/eaai48/revision/bench" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/eaai48/revision/bench")
endforeach()
